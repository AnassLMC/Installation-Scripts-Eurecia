📊 Calculatrice de Temps Badge
JavaScript 
// ==UserScript==
// @name         Eurecia Badge Time Left Calculator v8
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Shows time needed to complete 8 hours on Eurecia dashboard
// @author       You
// @match        https://plateforme.eurecia.com/eurecia/index.html*
// @grant        none
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    // Function to parse time string "HHhMM" or "HH:MM" format
    function parseTime(timeStr) {
        if (!timeStr) return 0;
        timeStr = timeStr.trim();
        const match = timeStr.match(/(\d+)[h:](\d+)/);
        if (match) {
            return parseInt(match[1]) * 60 + parseInt(match[2]); // return in minutes
        }
        return 0;
    }

    // Function to format minutes to "Xh XXmin"
    function formatTime(minutes) {
        const hours = Math.floor(minutes / 60);
        const mins = minutes % 60;
        return `${hours}h ${mins.toString().padStart(2, '0')}min`;
    }

    // Function to extract total worked hours and start time from modal
    function getWorkedHoursAndStartTime() {
        const allText = document.body.innerText;
        const lines = allText.split('\n');

        let workedHours = null;
        let startTime = null;

        // Extract total worked hours
        for (let i = 0; i < lines.length; i++) {
            if (lines[i].includes('Total') && i + 1 < lines.length) {
                const nextLine = lines[i + 1].trim();
                const timeMatch = nextLine.match(/(\d+)h(\d+)/);
                if (timeMatch) {
                    const hours = parseInt(timeMatch[1]);
                    const minutes = parseInt(timeMatch[2]);
                    workedHours = hours * 60 + minutes;
                }
            }

            // Extract LAST "Début" (most recent start time - after lunch return)
            if (lines[i].includes('Début') && i + 1 < lines.length) {
                const nextLine = lines[i + 1].trim();
                const timeMatch = nextLine.match(/(\d+)h(\d+)/);
                if (timeMatch) {
                    const hours = parseInt(timeMatch[1]);
                    const minutes = parseInt(timeMatch[2]);
                    // Keep updating - last one wins
                    startTime = { hours, minutes };
                }
            }
        }

        return { workedHours, startTime };
    }

    // Function to calculate end time (start + remaining work hours needed)
    function calculateEndTime(startTime, workedMinutes) {
        if (!startTime || workedMinutes === null) return null;

        const totalWorkNeeded = 8 * 60; // 8 hours in minutes
        const remainingMinutes = totalWorkNeeded - workedMinutes;

        // Add remaining work time to start time
        let endHours = startTime.hours;
        let endMinutes = startTime.minutes + remainingMinutes;

        // Handle minute overflow
        endHours += Math.floor(endMinutes / 60);
        endMinutes = endMinutes % 60;

        // Handle hour overflow (past midnight)
        if (endHours >= 24) {
            endHours -= 24;
        }

        return {
            hours: endHours,
            minutes: endMinutes,
            formatted: `${endHours.toString().padStart(2, '0')}:${endMinutes.toString().padStart(2, '0')}`
        };
    }

    // Function to create and inject the time remaining display
    function injectTimeDisplay(retryCount = 0) {
        // Find the info button
        const infoButton = document.querySelector('.total-hours-info-button');
        if (!infoButton) {
            console.log('Info button not found - retrying...');
            if (retryCount < 3) {
                setTimeout(() => injectTimeDisplay(retryCount + 1), 1000);
            }
            return;
        }

        // Click the info button
        infoButton.click();
        console.log('Info button clicked');

        // Aggressive polling with longer timeout
        let attempts = 0;
        const pollInterval = setInterval(() => {
            attempts++;

            const { workedHours, startTime } = getWorkedHoursAndStartTime();

            if (workedHours !== null && startTime) {
                clearInterval(pollInterval);
                console.log('Data found! Rendering...');
                console.log('Start time extracted:', startTime);
                console.log('Worked hours (minutes):', workedHours);

                const totalMinutes = 8 * 60;
                const remainingMinutes = Math.max(0, totalMinutes - workedHours);

                console.log('Remaining work minutes:', remainingMinutes);

                // Calculate day end time (start + remaining work hours)
                const endTime = calculateEndTime(startTime, workedHours);
                console.log('Calculated end time:', endTime);

                // Find the badgeuse section
                const badgeuseSection = document.querySelector('[data-v-598134f9].badgeuse-info');

                if (!badgeuseSection) {
                    console.log('Badge section not found');
                    return;
                }

                // Create the day end display element
                let timeDisplay = document.getElementById('eurecia-time-left-display');
                if (!timeDisplay) {
                    timeDisplay = document.createElement('div');
                    timeDisplay.id = 'eurecia-time-left-display';
                    timeDisplay.style.cssText = `
                        margin-top: 16px;
                        display: flex;
                        flex-direction: column;
                        gap: 4px;
                    `;

                    // Insert after the badge time section
                    const insertTarget = badgeuseSection.closest('.sub-content') || badgeuseSection.parentElement;
                    if (insertTarget) {
                        insertTarget.appendChild(timeDisplay);
                    }
                }

                // Update the display with day end time only - matching page styling
                timeDisplay.innerHTML = `
                    
Day end at

                    
${endTime.formatted}

                `;

                // Close the modal
                setTimeout(() => {
                    const closeButton = document.querySelector('[class*="close"]');
                    if (closeButton) {
                        closeButton.click();
                    }
                }, 200);
            } else if (attempts > 50) {
                // Timeout after 5 seconds (50 * 100ms)
                clearInterval(pollInterval);
                console.log('Timeout - retrying entire process');

                // Close any open modal
                const closeButton = document.querySelector('[class*="close"]');
                if (closeButton) {
                    closeButton.click();
                }

                // Retry if failed
                if (retryCount < 3) {
                    setTimeout(() => injectTimeDisplay(retryCount + 1), 1000);
                }
            }
        }, 100);
    }

    // Initial injection - try immediately, then fallback to load event
    function tryInitialInjection() {
        const badgeuseSection = document.querySelector('[data-v-598134f9].badgeuse-info');
        if (badgeuseSection) {
            // DOM is ready
            injectTimeDisplay();
        } else {
            // Wait for page to load
            window.addEventListener('load', () => {
                setTimeout(injectTimeDisplay, 300);
            });
        }
    }

    tryInitialInjection();

    // Also trigger on page visibility change (if user switches tabs and comes back)
    document.addEventListener('visibilitychange', () => {
        if (!document.hidden) {
            setTimeout(injectTimeDisplay, 500);
        }
    });

})();

// ==UserScript==
// @name         Eurecia PeapolDesk Activity Check Popup v4
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Shows popup when clicking "Je pars" to verify PeapolDesk activity stopped
// @author       You
// @match        https://plateforme.eurecia.com/eurecia/index.html*
// @grant        GM_addStyle
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    // Add CSS for popup and flashing background
    const popupCSS = `
        /* Flashing background overlay */
        .peapoldesk-overlay {
            position: fixed;
            top: 0;
            left: 0;
            right: 0;
            bottom: 0;
            z-index: 9999;
            animation: flash-background 0.8s infinite;
        }

        @keyframes flash-background {
            0%, 100% {
                background-color: rgba(0, 0, 0, 0.4);
            }
            50% {
                background-color: rgba(255, 0, 0, 0.4);
            }
        }

        /* Popup Modal */
        .peapoldesk-popup {
            position: fixed;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            z-index: 10000;
            background-color: white;
            border-radius: 8px;
            box-shadow: 0 8px 32px rgba(0, 0, 0, 0.3);
            padding: 32px;
            min-width: 400px;
            max-width: 500px;
            animation: slide-in 0.3s ease-out;
        }

        @keyframes slide-in {
            from {
                opacity: 0;
                transform: translate(-50%, -60%);
            }
            to {
                opacity: 1;
                transform: translate(-50%, -50%);
            }
        }

        /* Popup Title */
        .peapoldesk-popup-title {
            font-size: 18px;
            font-weight: 600;
            color: #333;
            margin-bottom: 16px;
            text-align: center;
        }

        /* Popup Message */
        .peapoldesk-popup-message {
            font-size: 16px;
            color: #555;
            margin-bottom: 28px;
            text-align: center;
            line-height: 1.6;
            font-weight: 500;
        }

        /* Popup Button Container */
        .peapoldesk-popup-buttons {
            display: flex;
            gap: 12px;
            justify-content: center;
        }

        /* OK Button */
        .peapoldesk-btn-ok {
            padding: 10px 28px;
            background-color: #1976d2;
            color: white;
            border: none;
            border-radius: 4px;
            font-size: 14px;
            font-weight: 600;
            cursor: pointer;
            transition: all 0.3s ease;
            text-transform: uppercase;
            letter-spacing: 0.5px;
        }

        .peapoldesk-btn-ok:hover {
            background-color: #1565c0;
            box-shadow: 0 4px 12px rgba(25, 118, 210, 0.4);
        }

        .peapoldesk-btn-ok:active {
            transform: scale(0.98);
        }

        /* Cancel Button (optional) */
        .peapoldesk-btn-cancel {
            padding: 10px 28px;
            background-color: #f5f5f5;
            color: #333;
            border: 1px solid #ddd;
            border-radius: 4px;
            font-size: 14px;
            font-weight: 600;
            cursor: pointer;
            transition: all 0.3s ease;
            text-transform: uppercase;
            letter-spacing: 0.5px;
        }

        .peapoldesk-btn-cancel:hover {
            background-color: #eeeeee;
            border-color: #bbb;
        }
    `;

    GM_addStyle(popupCSS);

    // Function to create and show the popup
    function showPeapolDeskPopup() {
        // Create overlay
        const overlay = document.createElement('div');
        overlay.className = 'peapoldesk-overlay';

        // Create popup
        const popup = document.createElement('div');
        popup.className = 'peapoldesk-popup';

        // Popup content
        popup.innerHTML = `
            
⚠️ Rappel Important

            

                Est ce que t'as stoppé l'activité sur PeapolDesk?
            

            

                
                
            

        `;

        // Add to page
        document.body.appendChild(overlay);
        document.body.appendChild(popup);

        // OK button handler
        document.getElementById('peapoldesk-ok').addEventListener('click', () => {
            overlay.remove();
            popup.remove();
            console.log('✅ PeapolDesk activity verified - Je pars action can continue');
        });

        // Cancel button handler
        document.getElementById('peapoldesk-cancel').addEventListener('click', () => {
            overlay.remove();
            popup.remove();
            console.log('❌ PeapolDesk activity check cancelled');
        });

        // Close on overlay click (optional)
        overlay.addEventListener('click', (e) => {
            if (e.target === overlay) {
                overlay.remove();
                popup.remove();
            }
        });
    }

    // Wait for page to load, then find and hook the "Je pars" button
    function setupJeParsBehavior() {
        const jePartButton = document.querySelector('[class*="timeclock-btn"]');

        if (jePartButton) {
            // Add click listener that shows popup BUT DOESN'T BLOCK the action
            jePartButton.addEventListener('click', () => {
                // Show popup as a warning (non-blocking)
                showPeapolDeskPopup();

                console.log('🔔 PeapolDesk check popup shown - Badge-out action continues');
            });

            console.log('✅ "Je pars" button hooked successfully');
            console.log('⚠️ Popup is a WARNING ONLY - badge-out will proceed regardless');
        } else {
            console.log('⚠️ "Je pars" button not found - retrying...');
            setTimeout(setupJeParsBehavior, 1000);
        }
    }

    // Initialize when page loads
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => {
            setTimeout(setupJeParsBehavior, 500);
        });
    } else {
        setTimeout(setupJeParsBehavior, 500);
    }

    console.log('%c🟠 PeapolDesk Activity Check Popup Script Loaded!', 'color: #ff9800; font-size: 14px; font-weight: bold;');

})();
